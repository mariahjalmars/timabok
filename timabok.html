const RATES = { 'Markaðsstofa Höfuðborgarsvæðisins': 18900, 'Markaðsstofa Norðurlands': 17900, 'Íslenski ferðaklasinn': 15900 }; function doGet(e) { const action = e.parameter.action; if (action === 'getMonth') { const month = e.parameter.month; const year = e.parameter.year; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheets()[0]; const data = sheet.getDataRange().getValues(); let totalHours = 0; let totalEarnings = 0; for (let i = 1; i < data.length; i++) { const dateCell = String(data[i][0]); if (dateCell.length === 10) { const rowMM = dateCell.substring(3, 5); const rowYYYY = dateCell.substring(6, 10); if (rowMM === month && rowYYYY === year) { const h = parseFloat(data[i][3]); if (!isNaN(h)) { totalHours += h; const company = String(data[i][1]); const rate = RATES[company] || 0; totalEarnings += h * rate; } } } } const today = new Date(); const lastDay = new Date(parseInt(year), parseInt(month), 0).getDate(); const daysLeft = (today.getMonth()+1 === parseInt(month) && today.getFullYear() === parseInt(year)) ? lastDay - today.getDate() : 0; return ContentService .createTextOutput(JSON.stringify({ totalHours: Math.round(totalHours * 10) / 10, totalEarnings: Math.round(totalEarnings), daysLeft: daysLeft })) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput('ok'); } function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = JSON.parse(e.postData.contents); sheet.appendRow([data.date, data.company, data.project, data.hours, data.note]); return ContentService.createTextOutput(JSON.stringify({status:"ok"})) .setMimeType(ContentService.MimeType.JSON); }
